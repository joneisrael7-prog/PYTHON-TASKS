store_a = {'LED': 20, 'Cable': 10, 'Board': 3}
store_b = {'Cable': 5, 'Sensor': 7, 'LED': 8}

combined = store_a.copy()

for item, quantity in store_b.items():
    if item in combined:
        combined[item] += quantity
    else:
        combined[item] = quantity

print("Combined inventory:", combined)

sales = {
    'Sensor': [4, 6, 5],
    'Cable': [10, 2, 3],
    'Board': [8, 7],
    'LED': [5, 5, 5]
}

totals = {}

for product, quantities in sales.items():
    totals[product] = sum(quantities)

ranked_products = sorted(
    totals.items(),
    key=lambda item: (-item[1], item[0])
)

for product, total in ranked_products:
    print(f"{product}: {total}")

prefix_dict = {0: 1}

for num in nums:
    prefix_sum += num

    if prefix_sum - k in prefix_dict:

    prefix_dict[prefix_sum] = prefix_dict.get(prefix_sum, 0) + 1

# 10.



from collections import Counter

s = 'ADOBECODEBANC'
t = 'ABC'

if not t or not s:
    result = ""
else:
    required = Counter(t)
    window = {}
    left = 0
    formed = 0
    required_chars = len(required)
    min_len = float('inf')
    start = 0

    for right, char in enumerate(s):
        window[char] = window.get(char, 0) + 1

        if char in required and window[char] == required[char]:
            formed += 1

        while formed == required_chars:
            if right - left + 1 < min_len:
                min_len = right - left + 1
                start = left

            left_char = s[left]
            window[left_char] -= 1

            if left_char in required and window[left_char] < required[left_char]:
                formed -= 1

            left += 1

    result = "" if min_len == float('inf') else s[start:start + min_len]

print("Minimum window:", repr(result))
