              #Caesar Cipher



message = input("Message: ")
shift = int(input("Shift: "))

encrypted = ""

# Encrypt the message
for ch in message:
    if 'A' <= ch <= 'Z':
        new_char = chr((ord(ch) - ord('A') + shift) % 26 + ord('A'))
        encrypted += new_char
    elif 'a' <= ch <= 'z':
        new_char = chr((ord(ch) - ord('a') + shift) % 26 + ord('a'))
        encrypted += new_char
    else:
        encrypted += ch

print("Encrypted:", encrypted)

decrypted = ""

# Decrypt the message
for ch in encrypted:
    if 'A' <= ch <= 'Z':
        new_char = chr((ord(ch) - ord('A') - shift) % 26 + ord('A'))
        decrypted += new_char
    elif 'a' <= ch <= 'z':
        new_char = chr((ord(ch) - ord('a') - shift) % 26 + ord('a'))
        decrypted += new_char
    else:
        decrypted += ch

print("Decrypted:", decrypted)




                 #Number Analyzer



start = int(input("Start: "))
end = int(input("End: "))

print(f"{'Number':<8}{'Prime':<8}{'Perfect':<10}"
      f"{'Armstrong':<12}{'Palindrome':<13}"
      f"{'Digit Sum':<12}{'Digits':<9}{'Binary'}")

for n in range(start, end + 1):
    # Prime check
    prime = True
    if n < 2:
        prime = False
    else:
        for i in range(2, n):
            if n % i == 0:
                prime = False
                break

    # Perfect number check
    divisor_sum = 0
    for i in range(1, n):
        if i > 0 and n % i == 0:
            divisor_sum += i
    perfect = n > 1 and divisor_sum == n

    # Digit sum, digit count and palindrome check
    temp = abs(n)
    digit_sum = 0
    digit_count = 0
    reverse = 0

    if temp == 0:
        digit_count = 1

    while temp > 0:
        digit = temp % 10
        digit_sum += digit
        digit_count += 1
        reverse = reverse * 10 + digit
        temp //= 10

    palindrome = n >= 0 and reverse == n

    # Armstrong number check
    temp = abs(n)
    armstrong_sum = 0

    while temp > 0:
        digit = temp % 10
        armstrong_sum += digit ** digit_count
        temp //= 10

    if n == 0:
        armstrong_sum = 0

    armstrong = n >= 0 and armstrong_sum == n

    # Binary representation
    if n < 0:
        binary = "-" + bin(abs(n))[2:]
    else:
        binary = bin(n)[2:]

    # Display row
    print(f"{n:<8}"
          f"{'Yes' if prime else 'No':<8}"
          f"{'Yes' if perfect else 'No':<10}"
          f"{'Yes' if armstrong else 'No':<12}"
          f"{'Yes' if palindrome else 'No':<13}"
          f"{digit_sum:<12}{digit_count:<9}{binary}")



               #Number System Converter + GCD/LCM



n1 = int(input("Number 1: "))
n2 = int(input("Number 2: "))

# Function to convert decimal to any base using loops
def convert(num, base):
    digits = "0123456789ABCDEF"
    result = ""

    while num > 0:
        remainder = num % base
        result = digits[remainder] + result
        num //= base

    return result if result else "0"

# Convert first number
print("Binary      :", convert(n1, 2))
print("Octal       :", convert(n1, 8))
print("Hexadecimal :", convert(n1, 16))

# Calculate GCD using Euclidean algorithm
a = n1
b = n2

while b != 0:
    remainder = a % b
    a = b
    b = remainder

gcd = a

# Calculate LCM
lcm = (n1 * n2) // gcd

print("GCD         :", gcd)
print("LCM         :", lcm)

                    #Student Mark Conversion & Result Analyzer


n = int(input("Students: "))

print(f"{'Student':<10}{'Total':<8}{'Average':<10}{'Grade'}")

for i in range(n):
    name = input("Student name: ")
    marks_input = input(f"{name} marks: ").split()

    try:
        marks = []
        for value in marks_input:
            mark = float(value)

            if mark < 0 or mark > 100:
                raise ValueError

            marks.append(mark)

        if len(marks) == 0:
            raise ValueError

        total = sum(marks)
        average = total / len(marks)

        if average >= 90:
            grade = "A+"
        elif average >= 80:
            grade = "A"
        elif average >= 70:
            grade = "B"
        elif average >= 60:
            grade = "C"
        elif average >= 50:
            grade = "D"
        else:
            grade = "F"

        print(f"{name:<10}{total:<8.0f}"
              f"{average:<10.2f}{grade}")

    except ValueError:
        print(f"{name:<10}Invalid marks")

                    #— String to Integer (atoi)



s = input("Enter a string: ")

i = 0
num = 0
sign = 1

# Skip leading whitespace
while i < len(s) and s[i].isspace():
    i += 1

# Check optional sign
if i < len(s) and (s[i] == '+' or s[i] == '-'):
    if s[i] == '-':
        sign = -1
    i += 1

# Convert consecutive digits
while i < len(s) and s[i].isdigit():
    digit = ord(s[i]) - ord('0')
    num = num * 10 + digit
    i += 1

# Print result
if i == 0 or (i == 1 and s[0] in '+-'):
    print(0)
else:
    print(sign * num)
