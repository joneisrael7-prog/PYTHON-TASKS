               #Run-Length Encoding of a List


values = [4, 4, 4, 2, 2, 7, 7, 4, 4, 9]

compressed = []
count = 1

for i in range(1, len(values)):
    if values[i] == values[i - 1]:
        count += 1
    else:
        compressed.append((values[i - 1], count))
        count = 1

compressed.append((values[-1], count))

print("Compressed list:", compressed)

    [7, 8],
    [9, 10],
    [11, 12]

    print("Error: Matrix dimensions are incompatible")
else:
    result = []

    for i in range(len(A)):
        row = []
        for j in range(len(B[0])):

            for k in range(len(B)):

        result.append(row)

    print("Product matrix:")

           # Maximum-Sum Path Through a Matrix


matrix = [
    [1, 3, 1],
    [2, 5, 2],
    [4, 2, 1]
]

rows = len(matrix)
cols = len(matrix[0])

dp = [[0] * cols for _ in range(rows)]
dp[0][0] = matrix[0][0]

for i in range(rows):
    for j in range(cols):
        if i == 0 and j == 0:
            continue
        top = dp[i - 1][j] if i > 0 else 0
        left = dp[i][j - 1] if j > 0 else 0
        dp[i][j] = matrix[i][j] + max(top, left)

path = []
i, j = rows - 1, cols - 1

while i > 0 or j > 0:
    path.append(matrix[i][j])

    if i == 0:
        j -= 1
    elif j == 0:
        i -= 1
    elif dp[i - 1][j] > dp[i][j - 1]:
        i -= 1
    else:
        j -= 1

path.append(matrix[0][0])
path.reverse()

print("Maximum path sum:", dp[rows - 1][cols - 1])
print("Path:", path)
